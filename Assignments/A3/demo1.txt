./vmsim --mode=bb --base=-4096 --limit=64 --trace=tests/bb/t1.txt

./vmsim --mode=bb --base=4096 --limit=64 --trace=missing.txt

./vmsim --mode=seg --config=missing.ini --trace=tests/seg/t2.txt
